class _GijacAppState extends ConsumerState<GijacApp> {
  StreamSubscription? _deepLinkSubscription;
  final CacheService _cacheService = CacheService();

  final ApiService _apiService = ApiService();

  @override
  void initState() {
    super.initState();
    _initDeepLinks();

  Future<void> _initPusherIfLoggedIn() async {
    debugPrint('🔍 Verificando sesión para Pusher...');
    try {
      final server = await _cacheService.getServerUrl();
      final token = await _cacheService.getDeviceToken();
      if (server == null || token == null) {
        debugPrint('⚠️ No hay sesión activa, no se conecta Pusher');
        return;
      }

      final profile = await _apiService.getProfile();
      final phoneNumberId = profile.phoneNumberId;
      debugPrint('🔍 phoneNumberId obtenido: $phoneNumberId'); // AGREGADO
      if (phoneNumberId == null) {
        debugPrint(
          '⚠️ phoneNumberId es null, no se conecta Pusher',
        ); // AGREGADO
        return;
      }

      debugPrint(
        '🔌 Conectando a Pusher, canal: presence-chat.$phoneNumberId',
      ); // AGREGADO

      await _pusherService.connect(
        server: server,
        deviceToken: token,
        phoneNumberId: phoneNumberId,
        onMensajeSent: (data) {
          final mensajeData = data['mensaje'] as Map<String, dynamic>?;
          if (mensajeData == null) return;

          final waFrom = mensajeData['wa_from']?.toString();
          final waTo = mensajeData['wa_to']?.toString();
          final myPhoneNumberId =
              data['phone_number_id']?.toString() ?? phoneNumberId;
          final bool esMio = waFrom == myPhoneNumberId;
          final String chatId = esMio ? (waTo ?? '') : (waFrom ?? '');
          if (chatId.isEmpty) return;

          final mapped = {
            'id': mensajeData['id'],
            'tipo': mensajeData['type'],
            'estado': mensajeData['estado'],
            'es mio': esMio,
            'metadata': mensajeData['metadata'],
            'contenido': mensajeData['body'],
            'fecha': mensajeData['sent_at'] ?? mensajeData['created_at'],
            'url_archivo': mensajeData['url_archivo'],
          };
          final message = Message.fromJson(mapped);

          final isChatOpen = ref
              .read(conversationProvider.notifier)
              .isViewingChat(chatId);
          ref
              .read(conversationProvider.notifier)
              .receiveRealtimeMessage(chatId, message);
          ref
              .read(chatsProvider.notifier)
              .receiveRealtimeMessage(
                chatId,
                message,
                isMine: esMio,
                isChatOpen: isChatOpen,
              );
        },
        onMensajeLeido: (data) {
          debugPrint('👁️ Mensaje leído: $data');
        },
      );
    } catch (e, st) {
      debugPrint('⚠️ No se pudo conectar Pusher al iniciar: $e');
      debugPrint('$st'); // AGREGADO: stack trace completo
    }
  }

  Future<void> _initDeepLinks() async {
    try {
      final uri = await _appLinks.getInitialLink();
      if (uri != null) {
        _handleDeepLink(uri);
      }
    } catch (e) {
      debugPrint('Error getting initial link: $e');
    }

    _deepLinkSubscription = _appLinks.uriLinkStream.listen(
      (Uri? uri) {
        if (uri != null) {
          _handleDeepLink(uri);
        }
      },
      onError: (err) {
        debugPrint('Error on deep link stream: $err');
      },
    );
  }

  Future<void> _handleDeepLink(Uri uri) async {
    debugPrint('Deep link received: $uri');

    if (uri.scheme == 'gijac' && uri.host == 'auth') {
      final token = uri.queryParameters['token'];
      final server = uri.queryParameters['server'];

      if (token != null && server != null) {
        // 1. Guardar servidor temporalmente
        await _cacheService.saveServerUrl(server);

        // 2. Obtener info del celular
        final deviceData = await _getDeviceInfo();

        // 3. Enviar datos a Laravel (no bloquea el flujo si falla)
        try {
          await _apiService.registerDevice({
            'token': token,
            'nombre_dispositivo': deviceData['nombre_dispositivo'],
            'modelo': deviceData['modelo'],
            'sistema_operativo': deviceData['sistema_operativo'],
            'version_so': deviceData['version_so'],
            'fcm_token': deviceData['fcm_token'],
            'ip': null,
          });
        } catch (e) {
          debugPrint('⚠️ No se pudo registrar dispositivo: $e');
        }

        // 4. Guardar el token (SIEMPRE)
        await _cacheService.saveDeviceToken(token);

        debugPrint('Token saved from deep link');

        _initPusherIfLoggedIn();

        if (mounted) {
          appRouter.go('/dashboard');
        }
      }
    }
  }

  Future<Map<String, dynamic>> _getDeviceInfo() async {
    final device = DeviceInfoPlugin();
    var info = {
      'nombre_dispositivo': 'Desconocido',
      'modelo': null,
      'sistema_operativo': 'Desconocido',
      'version_so': null,

    required this.name,
    required this.phone,
    this.about = 'Disponible',
    this.email,
    this.company,
    this.tags = const [],
    this.isOnline = false,
    this.avatarColorValue = 0xFF25D366,
  });

  String get initials {
    try {
      final parts = name.trim().split(RegExp(r'\s+'));
      if (parts.isEmpty || parts.first.isEmpty) return '?';
      if (parts.length == 1) return parts.first[0].toUpperCase();
      return (parts.first[0] + parts.last[0]).toUpperCase();
    } catch (e) {
      return '?';
    }
  }

  factory Contact.fromJson(Map<String, dynamic> json) {
    int colorValue = 0xFF25D366;
    final idStr = json['contact_id']?.toString() ?? json['id']?.toString() ?? '';
    if (idStr.isNotEmpty) {
      int hash = 0;
      for (int i = 0; i < idStr.length; i++) {
        hash += idStr.codeUnitAt(i);
      }
      final colors = [0xFF25D366, 0xFF128C7E, 0xFF3B82F6, 0xFF8B5CF6, 0xFFF59E0B, 0xFFEF4444, 0xFF06B6D4, 0xFFEC4899];
      colorValue = colors[hash % colors.length];
    }

    return Contact(
      id: json['contact_id']?.toString() ?? json['id']?.toString() ?? '',
      name: _cleanString(json['nombre'] ?? 'Sin nombre'),
      phone: json['telefono']?.toString() ?? '',
      about: 'Disponible',
      avatarColorValue: colorValue,
    );
  }

        'phone': phone,
        'about': about,
        'email': email,

        'isOnline': isOnline,
        'avatarColorValue': avatarColorValue,

  static String _cleanString(dynamic input) {
    if (input == null) return 'Sin datos';
    try {
      final bytes = utf8.encode(input.toString());
      String cleaned = utf8.decode(bytes, allowMalformed: true);
      cleaned = cleaned.replaceAll('�', '').trim();
      return cleaned.isEmpty ? 'Sin datos' : cleaned;
    } catch (e) {
      return 'Sin datos';
    }
  }

    if (json['hora_ultimo_mensaje'] != null) {
      try {
        final dateTime = DateTime.parse(json['hora_ultimo_mensaje'].toString());
        timeText = _formatTime(dateTime);
      } catch (e) {
        timeText = null;
      }
    }

    return Chat(
      id: json['id']?.toString() ?? json['numero_completo']?.toString() ?? '',
      contact: Contact.fromJson(json),
      lastMessage: _cleanString(json['ultimo_mensaje'] ?? 'Sin mensajes'),
      lastTime: DateTime.now(),
      unreadCount: int.tryParse(json['no_leidos']?.toString() ?? '0') ?? 0,
      rawTime: timeText != null ? _cleanString(timeText) : null,
    );
  }

          orElse: () => ChatStatus.open,

        isMuted: json['isMuted'] ?? false,
        assignedAgent: json['assignedAgent'],
        rawTime: json['rawTime'],

  static String _cleanString(dynamic input) {
    if (input == null) return 'Sin datos';
    try {
      final bytes = utf8.encode(input.toString());
      String cleaned = utf8.decode(bytes, allowMalformed: true);
      cleaned = cleaned.replaceAll('�', '').trim();
      return cleaned.isEmpty ? 'Sin datos' : cleaned;
    } catch (e) {
      return 'Sin datos';
    }
  }

  static String _formatTime(DateTime dateTime) {
    final now = DateTime.now();
    final difference = now.difference(dateTime);

    if (difference.inDays == 0) {
      if (difference.inMinutes < 1) return 'Ahora';
      if (difference.inHours < 1) return 'Hace ${difference.inMinutes} min';
      final hour = dateTime.hour > 12 ? dateTime.hour - 12 : (dateTime.hour == 0 ? 12 : dateTime.hour);
      final min = dateTime.minute.toString().padLeft(2, '0');
      final ampm = dateTime.hour >= 12 ? 'PM' : 'AM';
      return '$hour:$min $ampm';
    } else if (difference.inDays == 1) {
      return 'Ayer';
    } else {
      return '${dateTime.day}/${dateTime.month}/${dateTime.year}';
    }
  }

    final bool esMio = json['es mio'] ?? false;

    // 4. Extraer datos específicos del metadata de WhatsApp
    final dynamic rawMetadata = json['metadata'];
    final Map<String, dynamic> metadata = (rawMetadata is Map<String, dynamic>) ? rawMetadata : {};

    // Extraer texto (en WA interactivo, el texto a veces viene en el body del metadata)
    String? texto = json['contenido'];
    if (texto == null && metadata.containsKey('interactive')) {
      texto = metadata['interactive']?['body']?['text'];
    }

    // 5. MANEJO ESPECIAL PARA PLANTILLAS (Tipo 7) --- CORREGIDO: USAR "metadata" EN VEZ DE "meta"
    if (tipo == MessageType.template) {
      // Determinar qué hay en el header de la plantilla
      final int? tipoHeader = metadata['tipo_header'] is int 
          ? metadata['tipo_header'] 
          : int.tryParse(metadata['tipo_header']?.toString() ?? '');
      
      MessageType? headerType;
      if (tipoHeader == 2) headerType = MessageType.image;
      else if (tipoHeader == 3) headerType = MessageType.video;
      else if (tipoHeader == 4) headerType = MessageType.document;

      // Extraer botones (Laravel los guarda como string en la BD, hay que decodificarlos)
      List<MessageAction> templateButtons = [];
      final rawButtons = metadata['buttons'];
      if (rawButtons != null) {
        List<dynamic> btnList = [];
        if (rawButtons is String && rawButtons.trim().isNotEmpty) {
          try {
            btnList = jsonDecode(rawButtons);
          } catch (e) {
            btnList = [];
          }
        } else if (rawButtons is List) {
          btnList = rawButtons;
        }
            
        for (var btn in btnList) {
          templateButtons.add(MessageAction(

            description: btn['type'] == 'PHONE_NUMBER' ? 'Teléfono' : (btn['url'] ?? ''),

        }
      }

        type: tipo,

        time: DateTime.parse(json['fecha']),
        text: texto,
        status: estado,
        urlArchivo: json['url_archivo'],

      );

    if (!meta.containsKey('interactive')) return [];

    final List<MessageAction> actions = [];
    final type = interactive?['type'];

          description: row['description'],

      }

class ChatsNotifier extends Notifier<List<Chat>> {
  @override
  List<Chat> build() {
    // Disparamos la petición en segundo plano
    _fetchRealChats();
    return [];
  }

  Future<void> _fetchRealChats() async {
    final cacheService = CacheService();

    // 1. CARGAR DEL CACHÉ INMEDIATAMENTE
    final cachedChats = await cacheService.getCachedChats();
    if (cachedChats.isNotEmpty) {
      state = cachedChats;
    }

    // 2. INTENTAR TRAER DE LA API
    try {
      final api = ref.read(apiServiceProvider);
      final chats = await api.getChats();
      state = chats; // Actualiza la UI

      // 3. ACTUALIZAR CACHÉ
      await cacheService.cacheChats(chats);
    } catch (e) {
      debugPrint('❌ Error cargando chats reales: $e');
      // Si falla la API, se queda con los datos del caché (si los hay)
    }
  }

  Chat byId(String id) => state.firstWhere((c) => c.id == id);

  void receiveRealtimeMessage(
    String chatId,
    Message message, {
    required bool isMine,
    required bool isChatOpen,
  }) {
    final idx = state.indexWhere((c) => c.id == chatId);
    if (idx == -1)
      return; // el chat no está en la lista todavía (contacto nuevo)

    final c = state[idx];
    final newUnread = (isMine || isChatOpen) ? 0 : c.unreadCount + 1;

    state = [
      for (final ch in state)
        if (ch.id == chatId)
          Chat(
            id: ch.id,
            contact: ch.contact,
            lastMessage: message.text ?? ch.lastMessage,
            lastTime: message.time,
            unreadCount: newUnread,
            status: ch.status,
            isPinned: ch.isPinned,
            isMuted: ch.isMuted,
            assignedAgent: ch.assignedAgent,
            messages: ch.messages,
            rawTime: 'Ahora',
          )
        else
          ch,
    ];
  }

  void markRead(String id) {
    state = [
      for (final c in state)
        if (c.id == id)
          Chat(
            id: c.id,
            contact: c.contact,
            lastMessage: c.lastMessage,
            lastTime: c.lastTime,
            unreadCount: 0,
            status: c.status,
            isPinned: c.isPinned,
            isMuted: c.isMuted,
            assignedAgent: c.assignedAgent,
            messages: c.messages,
            rawTime: c.rawTime,

  switch (filter) {
    case ChatFilter.unread:
      result = result.where((c) => c.unreadCount > 0);
      break;
    case ChatFilter.bot:
      result = result.where((c) => c.status == ChatStatus.bot);
      break;
    case ChatFilter.resolved:
      result = result.where((c) => c.status == ChatStatus.resolved);
      break;
    case ChatFilter.all:
      break;
  }
  if (query.isNotEmpty) {
    result = result.where(
      (c) =>
          c.contact.name.toLowerCase().contains(query) ||
          c.lastMessage.toLowerCase().contains(query),
    );
  }
  final list = result.toList()
    ..sort((a, b) {
      if (a.isPinned != b.isPinned) return a.isPinned ? -1 : 1;
      return b.lastTime.compareTo(a.lastTime);
    });
  return list;
});

class ConversationNotifier extends Notifier<List<Message>> {
  @override
  List<Message> build() => [];

  bool _isLoadingMore = false;
  String? _activeChatId;
  final CacheService _cacheService = CacheService(); // <--- AGREGADO

  // AGREGADO: saber si el chat dado es el que está abierto ahora mismo
  bool isViewingChat(String chatId) => _activeChatId == chatId;

  // AGREGADO: insertar un mensaje que llegó por Pusher, solo si es el chat abierto
  void receiveRealtimeMessage(String chatId, Message message) {
    if (_activeChatId != chatId) return;
    if (state.any((m) => m.id == message.id)) return; // evitar duplicados
    state = [...state, message];
    _cacheService.cacheMessages(chatId, state);
  }

  /// Cargar mensajes al entrar (o reentrar) al chat
  Future<void> loadMessages(String numero) async {
    _activeChatId = numero;
    final loadingState = ref.read(isConversationLoadingProvider.notifier);
    loadingState.state = true;

    // 1. LEER DEL CACHÉ INMEDIATAMENTE (Estilo WhatsApp)
    final cachedMessages = await _cacheService.getCachedMessages(numero);
    if (_activeChatId != numero) return;
    if (cachedMessages.isNotEmpty) {
      state = cachedMessages;
      loadingState.state =
          false; // Ya tenemos algo que mostrar, quitamos el loading
    }

    // 2. INTENTAR TRAER DE LA API (En segundo plano)
    try {
      final api = ref.read(apiServiceProvider);
      final res = await api.getChatMessages(numero);
      if (_activeChatId != numero) return;

      final List<dynamic> rawMessages = res['messages'] ?? [];
      final apiMessages = rawMessages.map((e) => Message.fromJson(e)).toList();

      state = apiMessages; // Actualizamos la UI con los datos frescos

      // 3. ACTUALIZAR CACHÉ con lo más reciente
      await _cacheService.cacheMessages(numero, apiMessages);

      // 4. Marcar como leído en segundo plano
      api.markAsRead(numero);
      ref.read(chatsProvider.notifier).markRead(numero);
    } catch (e) {
      debugPrint('❌ Error cargando mensajes de API: $e');

      // Si NO HABÍA caché y falló la API, dejamos el estado vacío para que la UI muestre el error
      if (cachedMessages.isEmpty) {
        state = [];
      }
      // Si SÍ HABÍA caché, simplemente dejamos los mensajes del caché en el 'state'
      // y el usuario nunca se entera de que no hay internet.
    } finally {
      // Asegurarnos de quitar el loading en caso de que no hubiera caché
      if (_activeChatId == numero) {
        loadingState.state = false;
      }
    }
  }

  /// Cargar mensajes anteriores (Scroll hacia arriba)
  Future<void> loadOlderMessages(String numero) async {
    if (state.isEmpty || _isLoadingMore) return;
    _isLoadingMore = true;

    try {
      final api = ref.read(apiServiceProvider);
      final res = await api.getChatMessages(numero, beforeId: state.first.id);

      final List<dynamic> rawMessages = res['messages'] ?? [];
      if (rawMessages.isNotEmpty) {
        final olderMessages = rawMessages
            .map((e) => Message.fromJson(e))
            .toList();
        state = [...olderMessages, ...state];

        // Actualizar el caché con el historial completo ahora
        await _cacheService.cacheMessages(numero, state);
      }
    } catch (e) {
      debugPrint('❌ Error cargando mensajes antiguos: $e');
    } finally {
      _isLoadingMore = false;
    }
  }

  /// Enviar mensaje de texto
  Future<void> sendMessage(String numero, String text) async {
    if (text.trim().isEmpty) return;

    // 1. Mostrar inmediatamente en la UI (Optimistic UI)
    final tempMessage = Message.localTemp(text: text);
    state = [...state, tempMessage];

    try {
      final api = ref.read(apiServiceProvider);
      final realMessage = await api.sendMessage(numero, text);

      // 2. Reemplazar el temporal con el real de la BD
      state = [
        for (final m in state)
          if (m.id == tempMessage.id) realMessage else m,
      ];

      // 3. Actualizar la lista principal de chats
      ref.read(chatsProvider.notifier).updateLastMessage(numero, text);

      // 4. ACTUALIZAR EL CACHÉ con el nuevo mensaje
      await _cacheService.cacheMessages(numero, state);
    } catch (e) {
      debugPrint('❌ Error al enviar mensaje: $e');
      // Quitar el mensaje temporal si falló (y actualizar caché para no guardar el temp)
      state = state.where((m) => m.id != tempMessage.id).toList();
    }
  }
}

import 'dart:convert'; // AGREGAR ESTO
import 'package:flutter/foundation.dart';

class CacheService {
  static const String _tokenKey = 'device_token';
  static const String _serverKey = 'server_url';
  static const String _tokenExpiryKey = 'token_expiry';
  static const String _profileKey = 'user_profile';

  // --- Token y Server ---

  Future<void> saveDeviceToken(String token) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(_tokenKey, token);
    final expiry = DateTime.now()
        .add(const Duration(days: 30))
        .millisecondsSinceEpoch;
    await prefs.setInt(_tokenExpiryKey, expiry);
  }

  Future<String?> getDeviceToken() async {
    final prefs = await SharedPreferences.getInstance();
    final token = prefs.getString(_tokenKey);
    final expiry = prefs.getInt(_tokenExpiryKey);

    if (token == null) return null;

    if (expiry != null && DateTime.now().millisecondsSinceEpoch > expiry) {
      await clearDeviceData();
      return null;
    }

    return token;
  }

  Future<void> saveServerUrl(String serverUrl) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(_serverKey, serverUrl);
  }

  Future<String?> getServerUrl() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getString(_serverKey);
  }

  Future<bool> hasDeviceData() async {
    final token = await getDeviceToken();
    final server = await getServerUrl();
    return token != null && server != null;
  }

  Future<void> clearDeviceData() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove(_tokenKey);
    await prefs.remove(_serverKey);
    await prefs.remove(_tokenExpiryKey);
  }

  // --- Perfil ---

  Future<void> saveProfile(Map<String, dynamic> profileData) async {
    final prefs = await SharedPreferences.getInstance();

        onMensajeSent(jsonDecode(event.data));
      }),
    );
    _subs.add(

        onMensajeLeido(jsonDecode(event.data));
      }),

import 'package:flutter/widgets.dart';

import '../constants/app_constants.dart';

/// Simple responsive utilities for phone / tablet adaptation.
class Responsive {
  Responsive._();

  static bool isMobile(BuildContext context) =>
      MediaQuery.sizeOf(context).width < AppConstants.mobileMaxWidth;

  static bool isTablet(BuildContext context) {
    final width = MediaQuery.sizeOf(context).width;
    return width >= AppConstants.mobileMaxWidth &&
        width < AppConstants.tabletMaxWidth;
  }

  static bool isDesktop(BuildContext context) =>
      MediaQuery.sizeOf(context).width >= AppConstants.tabletMaxWidth;

  /// Returns a value based on the current breakpoint.
  static T value<T>(
    BuildContext context, {
    required T mobile,
    T? tablet,
    T? desktop,
  }) {
    if (isDesktop(context)) return desktop ?? tablet ?? mobile;
    if (isTablet(context)) return tablet ?? mobile;
    return mobile;
  }

  /// Horizontal content padding that grows on larger screens.
  static double horizontalPadding(BuildContext context) =>
      value(context, mobile: 16, tablet: 32, desktop: 64);
}

import 'package:intl/intl.dart';

/// Helpers for human-friendly time labels.
class TimeFormat {
  TimeFormat._();

  /// e.g. "14:32"
  static String clock(DateTime t) => DateFormat.Hm().format(t);

  /// Relative label used in lists: "ahora", "5 min", "3 h", "Ayer", "12/03".
  static String relative(DateTime t) {
    final now = DateTime.now();
    final diff = now.difference(t);
    if (diff.inMinutes < 1) return 'ahora';
    if (diff.inMinutes < 60) return '${diff.inMinutes} min';
    if (diff.inHours < 24 && now.day == t.day) return clock(t);
    if (diff.inHours < 48) return 'Ayer';
    return DateFormat('dd/MM').format(t);
  }

  /// Date divider label for conversations.
  static String dayLabel(DateTime t) {
    final now = DateTime.now();
    if (now.day == t.day && now.month == t.month && now.year == t.year) {
      return 'HOY';
    }
    final yesterday = now.subtract(const Duration(days: 1));
    if (yesterday.day == t.day && yesterday.month == t.month) return 'AYER';
    return DateFormat('d MMMM', 'es').format(t).toUpperCase();
  }
}

class _QrScannerScreenState extends ConsumerState<QrScannerScreen> {
  final CacheService _cacheService = CacheService();
  bool _isProcessing = false;
  MobileScannerController cameraController = MobileScannerController();

  void _onQRDetected(BarcodeCapture capture) {
    if (_isProcessing) return;

    final barcodes = capture.barcodes;
    if (barcodes.isEmpty) return;

    final code = barcodes.first.rawValue;
    if (code == null || code.isEmpty) return;

    setState(() => _isProcessing = true);

    _processQRData(code);
  }

    Future<void> _processQRData(String rawData) async {
    try {
      final uri = Uri.parse(rawData);

      String? token;
      String? server;

      if (uri.scheme == 'gijac') {
        token = uri.queryParameters['token'];
        server = uri.queryParameters['server'];
      } else if (uri.scheme == 'http' || uri.scheme == 'https') {
        token = uri.queryParameters['token'];
        server = uri.queryParameters['server'];
      } else {
        try {
          final json = jsonDecode(rawData);
          token = json['token'];
          server = json['server'];
        } catch (_) {}
      }

      if (token != null && server != null) {
        // 1. Guardar servidor temporalmente
        await _cacheService.saveServerUrl(server);
         await _cacheService.saveServerUrl(server);

        // 2. CONSULTAR Y GUARDAR EN CACHÉ EL PERFIL DEL USUARIO
        try {
          final api = ApiService();
          final repo = UserRepository(api, _cacheService);
          await repo.getProfile(); // Internamente lo guarda en la caché
          debugPrint('✅ Perfil guardado en caché tras escanear QR');
        } catch (e) {
          // Si falla (ej. sin internet), no bloqueamos el ingreso. 
          // El dashboard o el splash lo intentarán cargar después.
          debugPrint('⚠️ Sin internet al escanear, se usará caché anterior si existe: $e');
        }

        // 3. Obtener info del celular
        final deviceInfoPlugin = DeviceInfoPlugin();
        Map<String, dynamic> deviceData = {
          'nombre_dispositivo': 'Desconocido',
          'modelo': null,
          'sistema_operativo': 'Desconocido',
          'version_so': null,
          'fcm_token': null,
        };

        try {
          if (Platform.isAndroid) {
            final android = await deviceInfoPlugin.androidInfo;
            deviceData = {
              'token': token,
              'nombre_dispositivo': android.model ?? 'Android',
              'modelo': android.model,
              'sistema_operativo': 'Android',
              'version_so': android.version.release,
              'fcm_token': null,
              'ip': null,
            };
          } else if (Platform.isIOS) {
            final ios = await deviceInfoPlugin.iosInfo;
            deviceData = {
              'token': token,
              'nombre_dispositivo': ios.name ?? 'iPhone',
              'modelo': ios.model,
              'sistema_operativo': 'iOS',
              'version_so': ios.systemVersion,
              'fcm_token': null,
              'ip': null,
            };
          }

          try {
            deviceData['fcm_token'] = await FirebaseMessaging.instance.getToken();
          } catch (e) {
            debugPrint('⚠️ No se pudo obtener el token FCM: $e');
          }

          // 4. Enviar a Laravel
          await ApiService().registerDevice(deviceData);
        } catch (e) {
          debugPrint('⚠️ Error al registrar dispositivo: $e');
        }

        // 5. Guardar el token
        await _cacheService.saveDeviceToken(token);

        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('✅ Dispositivo vinculado exitosamente'),
              backgroundColor: Color(0xFF25D366),
              behavior: SnackBarBehavior.floating,
            ),
          );

          await Future.delayed(const Duration(milliseconds: 500));
          appRouter.go('/dashboard');
        }
      } else {
        _showError('Código QR no válido');
      }
    } catch (e) {
      _showError('Error al procesar el QR');
    } finally {
      if (mounted) {
        setState(() => _isProcessing = false);
      }
    }
  }

  void _showError(String message) {
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(message),
          backgroundColor: Colors.red,
          behavior: SnackBarBehavior.floating,
        ),
      );
    }
  }

  @override
  void dispose() {
    cameraController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.black,
      body: Stack(
        children: [
          MobileScanner(
            controller: cameraController,
            onDetect: _onQRDetected,
          ),

          Center(
            child: Container(
              width: 250,
              height: 250,
              decoration: BoxDecoration(
                borderRadius: BorderRadius.circular(24),
                border: Border.all(
                  color: const Color(0xFF25D366),
                  width: 3,
                ),
              ),
            ),
          ),

          if (_isProcessing)
            Container(
              color: Colors.black54,
              child: const Center(
                child: CircularProgressIndicator(color: Color(0xFF25D366)),
              ),
            ),

          Positioned(

    final lower = fileName.toLowerCase();
    if (lower.endsWith('.pdf')) return Icons.picture_as_pdf;
    if (lower.endsWith('.doc') || lower.endsWith('.docx')) return Icons.description;
    if (lower.endsWith('.xls') || lower.endsWith('.xlsx')) return Icons.table_chart;
    if (lower.endsWith('.ppt') || lower.endsWith('.pptx')) return Icons.slideshow;
    if (lower.endsWith('.zip') || lower.endsWith('.rar')) return Icons.folder_zip;
    return Icons.insert_drive_file;
  }

    if (fileName == null) return AppColors.blue;
    final lower = fileName.toLowerCase();
    if (lower.endsWith('.pdf')) return AppColors.danger;
    if (lower.endsWith('.doc') || lower.endsWith('.docx')) const Color(0xFF4CAF50); // Success Green
    if (lower.endsWith('.xls') || lower.endsWith('.xlsx')) return AppColors.blue;
    if (lower.endsWith('.ppt') || lower.endsWith('.pptx')) const Color(0xFFFF9800); // Warning Orange

class _AudioTile extends StatefulWidget {
  const _AudioTile({required this.message, required this.textColor});

class _AudioTileState extends State<_AudioTile> {
  final AudioPlayer _player = AudioPlayer();
  final CacheService _cacheService = CacheService();
  bool _isPlaying = false;
  bool _isLoading = false;
  Duration _duration = Duration.zero;
  Duration _position = Duration.zero;

  final List<double> _waveHeights = const [
    0.4, 0.7, 0.5, 0.9, 0.6, 0.8, 0.45, 0.75, 0.55, 0.85,
    0.5, 0.7, 0.4, 0.65, 0.55, 0.75, 0.45, 0.8, 0.6, 0.5
  ];

  @override
  void initState() {
    super.initState();
    _player.playerStateStream.listen((state) {
      if (!mounted) return;
      setState(() => _isPlaying = state.playing);
      if (state.processingState == ProcessingState.completed) {
        _player.seek(Duration.zero);
        _player.pause();
      }
    });
    _player.durationStream.listen((d) {
      if (d != null && mounted) setState(() => _duration = d);
    });
    _player.positionStream.listen((p) {
      if (mounted) setState(() => _position = p);
    });
  }

  Future<void> _togglePlay() async {
    try {
      if (_player.audioSource == null) {
        setState(() => _isLoading = true);
        final path = widget.message.urlArchivo;
        if (path == null) return;

        String fullUrl = path;
        if (!path.startsWith('http')) {
          final server = await _cacheService.getServerUrl();
          fullUrl = '$server$path';
        }

        await _player.setUrl(fullUrl);
        setState(() => _isLoading = false);
      }

      if (_isPlaying) {
        await _player.pause();
      } else {
        await _player.play();
      }
    } catch (e) {
      debugPrint('❌ Error reproduciendo audio: $e');
      setState(() => _isLoading = false);
    }
  }

  String _formatDuration(Duration d) {
    final minutes = d.inMinutes.remainder(60).toString();
    final seconds = d.inSeconds.remainder(60).toString().padLeft(2, '0');
    return '$minutes:$seconds';
  }

  @override
  void dispose() {
    _player.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final progress = _duration.inMilliseconds > 0

class _TemplateContent extends StatelessWidget {
  const _TemplateContent({required this.message, required this.textColor});
  final Message message;
  final Color textColor;

  @override
  Widget build(BuildContext context) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        // 1. HEADER (Imagen, Video o Documento)
        if (message.templateHeaderUrl != null) ...[
          _buildHeader(context),
          const SizedBox(height: 8),
        ],
        
        // 2. TEXTO (Body)
        if (message.text != null)
          Padding(
            padding: const EdgeInsets.only(bottom: 8),
            child: Text(message.text!, style: TextStyle(color: textColor, height: 1.35)),
          ),

        // 3. BOTONES
        if (message.actions.isNotEmpty)
          ...message.actions.map((a) => Container(
                width: 240,
                margin: const EdgeInsets.only(top: 4),
                padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                decoration: BoxDecoration(
                  color: Colors.black.withValues(alpha: 0.04),
                  borderRadius: BorderRadius.circular(8),
                ),
                child: Row(
                  children: [
                    Icon(
                      a.description == 'Teléfono' ? Icons.phone_outlined : Icons.open_in_new,
                      size: 18, 
                      color: AppColors.secondary,
                    ),
                    const SizedBox(width: 10),
                    Expanded(
                      child: Text(a.label, style: TextStyle(color: textColor, fontWeight: FontWeight.w600)),
                    ),
                  ],
                ),
              )),
      ],
    );
  }

  Widget _buildHeader(BuildContext context) {
    switch (message.templateHeaderType) {
      case MessageType.image:
        return ClipRRect(
          borderRadius: BorderRadius.circular(10),
          child: Image.network(
            message.templateHeaderUrl!,
            fit: BoxFit.cover,
            loadingBuilder: (context, child, progress) {
              if (progress == null) return child;
              return const Center(child: CircularProgressIndicator(strokeWidth: 2));
            },
            errorBuilder: (_, __, ___) => const Icon(Icons.broken_image, size: 50),
          ),
        );
      case MessageType.video:
        // En plantillas, el "video" suele ser solo una imagen de portada
        return ClipRRect(
          borderRadius: BorderRadius.circular(10),
          child: Stack(
            alignment: Alignment.center,
            children: [
              Image.network(message.templateHeaderUrl!, fit: BoxFit.cover, errorBuilder: (_, __, ___) => const SizedBox()),
              Container(color: Colors.black26, child: const Icon(Icons.play_circle_fill, size: 50, color: Colors.white70)),
            ],
          ),
        );
      case MessageType.document:
        // Reutilizamos la lógica de archivo
        return _FileTile(
          message: Message(
            id: '', 
            type: MessageType.document, 
            source: MessageSource.agent, 
            time: DateTime.now(), 
            urlArchivo: message.templateHeaderUrl,
            fileName: message.templateHeaderUrl?.split('/').last,
          ), 
          textColor: textColor
        );
      default:
        return const SizedBox.shrink();
    }
  }
}

        body: Row(
          children: [
            NavigationRail(
              selectedIndex: index,
              onDestinationSelected: onTap,
              labelType: NavigationRailLabelType.all,
              destinations: [
                for (final d in _destinations)
                  NavigationRailDestination(
                    icon: _withBadge(
                        Icon(d.$2), d.$1 == Routes.notifications ? unread : 0),
                    selectedIcon: _withBadge(Icon(d.$3),
                        d.$1 == Routes.notifications ? unread : 0),
                    label: Text(d.$4),
                  ),
              ],
            ),
            const VerticalDivider(width: 1),
            Expanded(child: child),
          ],
        ),
      );
    }

    return Scaffold(
      body: child,
      bottomNavigationBar: NavigationBar(
        selectedIndex: index,
        onDestinationSelected: onTap,
        destinations: [
          for (final d in _destinations)
            NavigationDestination(
              icon: _withBadge(
                  Icon(d.$2), d.$1 == Routes.notifications ? unread : 0),
              selectedIcon: _withBadge(
                  Icon(d.$3), d.$1 == Routes.notifications ? unread : 0),
              label: d.$4,
            ),
        ],
